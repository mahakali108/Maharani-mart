create or replace function public.hold_coupon_evidence_delete()
returns trigger language plpgsql set search_path = pg_catalog, pg_temp as $$
begin
  raise exception 'Coupon evidence deletion is held (%).', tg_table_name
    using errcode = '23514';
  return old;
end;
$$;
drop trigger if exists trg_hold_coupon_evidence_delete on public.coupon_redemptions;
create trigger trg_hold_coupon_evidence_delete
  before delete on public.coupon_redemptions
  for each row execute function public.hold_coupon_evidence_delete();

  before delete on public.coupons

revoke delete, truncate on table public.coupons, public.coupon_redemptions
